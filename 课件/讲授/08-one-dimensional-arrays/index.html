14 · 完整程序

遍历数组并输出最大值

#include <stdio.h>

int main()
{
    double a[10];
    double maximum;
    int i;
    for (i = 0; i < 10; i++)
    {
        scanf("%lf", &a[i]);
    }
    maximum = a[0];
    for (i = 1; i < 10; i++)
    {
        if (a[i] > maximum)
        {
            maximum = a[i];
        }
    }
    printf("%.6f\n", maximum);
    return 0;
}
输入约束：程序运行时提供恰好10个合法实数。

24 · 完整程序

十个整数冒泡排序

#include <stdio.h>

int main()
{
    int a[10];
    int i, j, t;
    for (i = 0; i < 10; i++)
    {
        scanf("%d", &a[i]);
    }
    for (j = 0; j < 9; j++)
    {
        for (i = 0; i < 9 - j; i++)
        {
            if (a[i] > a[i + 1])
            {
                t = a[i];
                a[i] = a[i + 1];
                a[i + 1] = t;
            }
        }
    }
    for (i = 0; i < 10; i++)
    {
        if (i > 0)
        {
            printf(" ");
        }
        printf("%d", a[i]);
    }
    printf("\n");
    return 0;
}
输入约束：程序运行时提供恰好10个合法整数。
