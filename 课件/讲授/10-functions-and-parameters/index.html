27 · 完整程序

调用排序子函数升序输出10个浮点数

#include <stdio.h>

void sort(double a[], int n);

int main()
{
    double a[10];
    int i;

    for (i = 0; i < 10; i++)
    {
        scanf("%lf", &a[i]);
    }

    sort(a, 10);

    for (i = 0; i < 10; i++)
    {
        if (i > 0)
        {
            printf(" ");
        }

        printf("%.2f", a[i]);
    }

    printf("\n");
    return 0;
}

void sort(double a[], int n)
{
    int i, j;
    double t;

    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - 1 - i; j++)
        {
            if (a[j] > a[j + 1])
            {
                t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
        }
    }
}
输入约束：程序运行时提供恰好10个合法实数。
