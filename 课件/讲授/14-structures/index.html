03 · 一条完整记录

相关数据应作为一个整体保存

分散的变量

int id;
char name[20];
double score;

变量之间没有显式标明“属于同一名学生”。

结构体记录

struct Student
{
    int id;
    char name[20];
    double score;
};

三个成员共同描述一名学生。

结构体把具有逻辑联系、但类型可以不同的数据组合成一条记录。

15 · 比较成绩成员

三个互斥结果分别对应不同输出

>

第一名成绩较高，输出student1。

<

第二名成绩较高，输出student2。

==

成绩相同，按输入顺序输出两条记录。

比较依据是成绩成员，输出对象仍是包含学号、姓名和成绩的完整记录。

16 · 完整程序

比较两条学生记录

#include <stdio.h>

struct Student
{
    int id;
    char name[20];
    double score;
};

int main()
{
    struct Student student1;
    struct Student student2;

    scanf("%d %19s %lf", &student1.id, student1.name, &student1.score);
    scanf("%d %19s %lf", &student2.id, student2.name, &student2.score);

    if (student1.score > student2.score)
    {
        printf("%d %s %.1f\n", student1.id, student1.name, student1.score);
    }
    else if (student1.score < student2.score)
    {
        printf("%d %s %.1f\n", student2.id, student2.name, student2.score);
    }
    else
    {
        printf("%d %s %.1f\n", student1.id, student1.name, student1.score);
        printf("%d %s %.1f\n", student2.id, student2.name, student2.score);
    }

    return 0;
}

26 · 降序比较条件

左侧成绩较小时交换两条记录

if (students[i].score
    < students[i + 1].score)
{
    /* 交换完整记录 */
}

条件为假

左侧成绩不小于右侧，顺序符合降序要求。

条件为真

右侧成绩更高，应把右侧完整记录移到左边。

29 · 为什么不能只交换成绩

成员被拆开后，记录表达的对象就会改变

错误：只交换成绩

学号290.0
学号375.0

成绩错误地归到了另一名学生。

正确：交换完整记录

学号390.0
学号275.0

每个学号仍与原成绩对应。

排序改变记录次序，不改变记录内部成员之间的关系。

32 · 完整程序

按成绩降序排列5条完整记录

#include <stdio.h>

struct Student
{
    int id;
    double score;
};

int main()
{
    struct Student students[5];
    struct Student temp;
    int i, j;

    for (i = 0; i < 5; i++)
    {
        scanf("%d %lf", &students[i].id, &students[i].score);
    }

    for (j = 0; j < 4; j++)
    {
        for (i = 0; i < 4 - j; i++)
        {
            if (students[i].score < students[i + 1].score)
            {
                temp = students[i];
                students[i] = students[i + 1];
                students[i + 1] = temp;
            }
        }
    }

    for (i = 0; i < 5; i++)
    {
        printf("%d %.1f\n", students[i].id, students[i].score);
    }

    return 0;
}
